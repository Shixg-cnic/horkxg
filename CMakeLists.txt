cmake_minimum_required(VERSION 3.24)

if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES native CACHE STRING "CUDA architectures")
endif()
project(gpart LANGUAGES CXX CUDA)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
find_package(CUDAToolkit REQUIRED)
include(CTest)

add_library(gpart_options INTERFACE)
target_include_directories(gpart_options INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(gpart_options INTERFACE
    $<$<COMPILE_LANGUAGE:CXX>:-O3;-march=native>
    $<$<COMPILE_LANGUAGE:CUDA>:-O3;-Xcompiler=-march=native>)
target_link_libraries(gpart_options INTERFACE CUDA::cudart_static)
set_source_files_properties(src/graph.cpp PROPERTIES LANGUAGE CUDA)

function(add_graph_variant suffix definition)
    # Reuse each compiled algorithm across drivers and tests of the same width.
    add_library(gpart_graph_${suffix} STATIC src/graph.cpp)
    target_link_libraries(gpart_graph_${suffix} PUBLIC gpart_options)
    target_compile_definitions(gpart_graph_${suffix} PUBLIC ${definition})
    add_library(gpart_coarsening_${suffix} STATIC src/coarsen.cu)
    target_link_libraries(gpart_coarsening_${suffix} PUBLIC gpart_graph_${suffix})
    add_library(gpart_initial_gpu_${suffix} STATIC src/initial_partition_gpu.cu)
    target_link_libraries(gpart_initial_gpu_${suffix} PUBLIC gpart_graph_${suffix})
    add_library(gpart_refinement_${suffix} STATIC src/refine.cu src/uncoarsen.cu)
    target_link_libraries(gpart_refinement_${suffix} PUBLIC gpart_graph_${suffix})
    add_executable(gpart_coarsen_${suffix} app/coarsen_main.cu)
    target_link_libraries(gpart_coarsen_${suffix} PRIVATE gpart_coarsening_${suffix})
    add_executable(gpart_partition_${suffix} app/partition_main.cu)
    target_link_libraries(gpart_partition_${suffix} PRIVATE
        gpart_coarsening_${suffix} gpart_initial_gpu_${suffix} gpart_refinement_${suffix})
    if(BUILD_TESTING)
        add_executable(gpart_test_uncoarsen_${suffix} tests/test_uncoarsen.cu)
        target_link_libraries(gpart_test_uncoarsen_${suffix} PRIVATE
            gpart_coarsening_${suffix} gpart_refinement_${suffix})
        add_test(NAME uncoarsen_small_${suffix} COMMAND gpart_test_uncoarsen_${suffix})
        add_executable(gpart_test_initial_gpu_${suffix} tests/test_initial_partition_gpu.cu)
        target_link_libraries(gpart_test_initial_gpu_${suffix} PRIVATE gpart_initial_gpu_${suffix})
        add_test(NAME gpart_test_initial_gpu_${suffix} COMMAND gpart_test_initial_gpu_${suffix})
    endif()
endfunction()
add_graph_variant(standard GPART_STANDARD_GRAPH)
add_graph_variant(big GPART_BIG_GRAPH)

if(BUILD_TESTING)
    add_executable(gpart_test_sclp_record tests/test_sclp_record.cpp)
    target_include_directories(gpart_test_sclp_record PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    add_test(NAME sclp_record_host COMMAND gpart_test_sclp_record)
    add_executable(gpart_test_partition_capacity tests/test_partition_capacity.cpp)
    target_include_directories(gpart_test_partition_capacity PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    add_test(NAME partition_capacity_host COMMAND gpart_test_partition_capacity)
endif()
